cmake_minimum_required(VERSION 3.22.1)
project(soundlink_opus)

set(OPUS_SOURCE_DIR ${CMAKE_CURRENT_SOURCE_DIR}/opus)

if(NOT EXISTS ${OPUS_SOURCE_DIR}/CMakeLists.txt)
    message(FATAL_ERROR "libopus source is required at ${OPUS_SOURCE_DIR}. Place the extracted opus source directory there before building Android native code.")
endif()

set(OPUS_INSTALL_PKG_CONFIG_MODULE OFF CACHE BOOL "" FORCE)
set(OPUS_INSTALL_CMAKE_CONFIG_MODULE OFF CACHE BOOL "" FORCE)
set(OPUS_BUILD_PROGRAMS OFF CACHE BOOL "" FORCE)
set(OPUS_BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(OPUS_BUILD_SHARED_LIBRARY OFF CACHE BOOL "" FORCE)
set(OPUS_BUILD_FRAMEWORK OFF CACHE BOOL "" FORCE)
set(OPUS_DISABLE_INTRINSICS ON CACHE BOOL "" FORCE)
set(OPUS_X86_MAY_HAVE_SSE OFF CACHE BOOL "" FORCE)
set(OPUS_X86_MAY_HAVE_SSE2 OFF CACHE BOOL "" FORCE)
set(OPUS_X86_MAY_HAVE_SSE4_1 OFF CACHE BOOL "" FORCE)
set(OPUS_X86_MAY_HAVE_AVX2 OFF CACHE BOOL "" FORCE)
set(OPUS_X86_PRESUME_SSE OFF CACHE BOOL "" FORCE)
set(OPUS_X86_PRESUME_SSE2 OFF CACHE BOOL "" FORCE)
set(OPUS_X86_PRESUME_SSE4_1 OFF CACHE BOOL "" FORCE)
set(OPUS_X86_PRESUME_AVX2 OFF CACHE BOOL "" FORCE)

add_subdirectory(${OPUS_SOURCE_DIR} ${CMAKE_CURRENT_BINARY_DIR}/opus-build)

add_library(soundlink_opus SHARED opus_jni.c)

target_link_libraries(soundlink_opus PRIVATE opus log)

target_include_directories(soundlink_opus PRIVATE
    ${OPUS_SOURCE_DIR}/include)
